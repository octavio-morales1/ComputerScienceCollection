a = np.array(np.random.random((10,10)))
print(a)
maxSum= a[0][0]
print(maxSum)
bestSquare= 0

for x in range(8):
  for y in range(8):
    square= a[x:x+3,y:y+3]
    total= square.sum()
    if (total> maxSum):
      maxSum= total
      bestSquare=square

print()
print("Maximum 3x3 Sub-Array")
print(bestSquare)
